/*
========================================================
 OPENEDUCA
 MAPA INTERACTIVO DEL LICEO ANTONIO VARAS DE LA BARRA
========================================================
*/

const ALTO_MAPA = 1000;
const ANCHO_MAPA = 600;
const limites = [[0, 0], [ALTO_MAPA, ANCHO_MAPA]];

// Inicialización del mapa en plano 2D
const mapa = L.map("mapa", {
    crs: L.CRS.Simple,
    minZoom: -1,
    maxZoom: 2,
    zoomControl: true,
    attributionControl: false,
    maxBoundsViscosity: 1.0
});

mapa.fitBounds(limites);
mapa.setMaxBounds(limites);

// Estilo unificado para los muros del plano
const estiloPared = {
    color: "#000000",
    weight: 4,
    lineCap: "round",
    lineJoin: "round"
};

/*
========================================================

========================================================
*/

for (let y = 210; y <= 420; y += 35) {

/*
========================================================
 FUNCIÓN CREADORA DE PUNTOS
========================================================
*/
function crearPunto(nombre, icono, posicion, color, descripcion) {
    const punto = L.circleMarker(posicion, {
        radius: 13,
        color: "#ffffff",
        weight: 2,
        fillColor: color,
        fillOpacity: 1
    }).addTo(mapa);

    punto.bindTooltip(`${icono} ${nombre}`, {
        permanent: false,
        direction: "top",
        sticky: true
    });

    punto.bindPopup(`
        <div style="min-width:180px">
            <strong style="font-size:16px;">${icono} ${nombre}</strong>
            <br><br>
            ${descripcion}
        </div>
    `);

    punto.on("mouseover", function() { this.setRadius(17); });
    punto.on("mouseout", function() { this.setRadius(13); });

    return punto;
}

========================================================

crearPunto("Administración", "📍", [230, 260], "#e74c3c", "Sector administrativo central.");

/*
========================================================
 COMPORTAMIENTO Y NAVEGACIÓN
========================================================
*/
mapa.on("drag", function() {
    mapa.panInsideBounds(limites, { animate: false });
});

mapa.on("popupopen", function(e) {
    const elemento = e.popup.getElement();
    if (elemento) {
        elemento.style.animation = "aparecerPopup .2s ease";
    }
});
